\begin{abstract}
A bare lag $N(t-g)$ admits adult biomass at time $t$ without a juvenile draw on $A$ during $[t-g,t]$. The core of this paper is the two-stage mass block $(X_A,X_J)$ with donor-limited Ricker birth, institutional memory, and the companion's gated effort law. Two harvest primitives are distinguished. Adult take removes $qEX_A$ from $X_A$. Juvenile take removes $qEX_J$ from $X_J$ and is a genuine mass flux, not the companion's prevented-recruitment floor. At Candidate~A effort parameters both interiors exist. Adult take is delay-independently stable at default $\eta$ and has a stabilising Hopf under elevated forcing ($\eta=5$, $q=0.01$, $g=5$: $\tau^*\approx 52.07$~yr, period $269.4$~yr, $\mathrm{Re}\,\lambda'(\tau^*)<0$). Juvenile take has a Hurwitz undelayed Jacobian and no imaginary-axis root on the searched set. The two-crossing diagram of~\cite{paperI} is not recovered.
\end{abstract}

\section{The core}

Frozen active pool $s=A/(A+A_0)\in(0,1]$, maturation time $g>0$, Ricker birth $B=P_0 X_A e^{-X_A/N_c}s$. Memory and effort are the companion's~\cite{paperI}: $Z^*=\delta$, $E^*$ the same positive root of the effort quadratic.

\begin{definition}[Adult take]
\label{def:adult}
\begin{equation}
\begin{aligned}
\dot X_A&=\frac1g X_J-d_A X_A-qEX_A,\\
\dot X_J&=B-\frac1g X_J-d_J X_J,
\end{aligned}
\label{eq:adult}
\end{equation}
with deficit $\phi=-\dot X_A$ driving $Z$.
\end{definition}

\begin{definition}[Juvenile take]
\label{def:juv}
\begin{equation}
\begin{aligned}
\dot X_A&=\frac1g X_J-d_A X_A,\\
\dot X_J&=B-\frac1g X_J-d_J X_J-qEX_J,
\end{aligned}
\label{eq:juv}
\end{equation}
with deficit $\phi=-\dot X_J$. Harvest $qEX_J$ is a ledger flux from $X_J$ to product or waste.
\end{definition}

The mixture $C_{\mathrm{recruit}}=(1-\psi)qEN$ of~\cite{paperI} suppresses birth and transfers no mass. Definition~\ref{def:juv} routes harvest as a flux from $X_J$.

\begin{theorem}[Mass and invariance]
\label{thm:mass}
Along~\eqref{eq:adult}, $\frac{d}{dt}(X_A+X_J)=-d_A X_A-d_J X_J-qEX_A$. Along~\eqref{eq:juv}, $\frac{d}{dt}(X_A+X_J)=-d_A X_A-d_J X_J-qEX_J$. The nonnegative orthant is forward invariant. Restoring $A$, $U$, and $P$ with the companion's donor-limited routing closes the ledger.
\end{theorem}

\begin{proof}
Adding~\eqref{eq:adult} gives
\[
\dot X_A+\dot X_J=B-d_A X_A-d_J X_J-qEX_A,
\]
and $B$ does not appear as a sink, so $\frac{d}{dt}(X_A+X_J)=-d_A X_A-d_J X_J-qEX_A$. Adding~\eqref{eq:juv} likewise yields $-d_A X_A-d_J X_J-qEX_J$. On $X_A=0$ every adult-take term vanishes and $\dot X_A=X_J/g\ge 0$; on $X_J=0$, $\dot X_J=B\ge 0$ (adult take) or $\dot X_J=B\ge 0$ (juvenile take, the harvest term being $qEX_J=0$). Nagumo's criterion \cite{nagumo1942} gives invariance of the closed orthant. Restoring $A$, $U$, and $P$ with the routing of~\cite{paperI} places each harvested unit in $P$ or $W$ and returns decay to $A$.
\end{proof}

\begin{theorem}[Interior equilibrium]
\label{thm:eq}
At $Z^*=\delta$ and the companion $E^*>0$, write
\[
\Theta_{\mathrm{ad}}=(d_A+qE^*)(1+gd_J),\qquad
\Theta_{\mathrm{juv}}=d_A\bigl(1+g(d_J+qE^*)\bigr).
\]
If $P_0 s>\Theta$, there is a unique positive stage rest
\[
X_A^*=N_c\ln(P_0 s/\Theta),\qquad
X_J^*=g X_A^*\cdot
\begin{cases}
d_A+qE^* & \text{adult take},\\
d_A & \text{juvenile take}.
\end{cases}
\]
\end{theorem}

\begin{proof}
Set the stage derivatives to zero and substitute $B=P_0 X_A e^{-X_A/N_c}s$. The displayed $\Theta$ is $B/X_A$.
\end{proof}

\section{Local spectrum}

The linearisation about Theorem~\ref{thm:eq} is a $4\times 4$ single-delay RFDE. Imaginary roots are zeros of $\det\Delta(i\omega,\tau)=0$ with $\Delta(\lambda)=\lambda I-A_0-A_\tau e^{-\lambda\tau}$.

\begin{numresult}[Computed crossings]
\label{num:hopf}
Baseline life-history: $P_0=1.2$, $N_c=50$, $d_A=0.15$, $d_J=0.4$, $s=0.99$, companion Candidate~A $(\eta,E_{\max},\delta_0,\Delta_{\mathrm{ref}},Z_{\mathrm{ref}},\tau_m,k)$. Residuals $|\det\Delta|<10^{-16}$ at reported points.

Harvest & Forcing & $X_A^*$ & First $\tau^*$ (yr) & Period (yr) \\
\midrule
Adult & $\eta=0.914$, $q=0.001$, $g=5$ & $47.85$ & none & --- \\
Adult & $\eta=5$, $q=0.01$, $g=5$ & $42.04$ & $52.068$ & $269.37$ \\
Adult & $\eta=5$, $q=0.01$, $g=1$ & $80.15$ & $16.757$ & $105.05$ \\

At the elevated $g=5$ adult crossing, $\mathrm{Re}\,\lambda'(\tau^*)=-1.54\times 10^{-4}<0$ (stabilising) and the undelayed Jacobian has two positive real eigenvalues ($0.243$, $0.039$). Juvenile take at the same elevated point is Hurwitz at $\tau=0$ (spectrum $-1.135$, $-0.149$, $-0.0047\pm 0.182\,i$).
\end{numresult}

Adult take is delay-stable at Candidate~A effort parameters. Under elevated forcing it has a single stabilising Hopf. Juvenile take produces no imaginary-axis root on the same set.

\begin{proposition}[Absence of a two-crossing spectrum]
\label{prop:not}
No searched adult or juvenile parameterisation produced a pair of Hopf delays with opposite crossing directions. Definitions~\ref{def:adult}--\ref{def:juv} therefore do not reproduce the five-regime diagram of~\cite{paperI}.
\end{proposition}

\section{Conclusion}

Adult take under elevated forcing has one stabilising Hopf. Juvenile take is locally delay-stable on the searched set. The delays $\tau_\pm$ of~\cite{paperI} are those of the unstructured logistic core.
