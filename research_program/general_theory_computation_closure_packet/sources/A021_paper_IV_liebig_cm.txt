\begin{abstract}
The companion core is a scalar specialisation. This paper supplies a conditional reduction from a smooth vector Liebig system to that core, under a uniform yield gap and an explicit spectral-gap hypothesis for the slack fibres. Off-limiting softmax weights are $O(e^{-\rho\Delta_y})$. The result is a local $C^1$ attracting invariant graph for a weakly coupled product system, with error $O(e^{-\rho\Delta_y})+O(\varepsilon_{\mathrm{phys}})$ on compact time; it is not claimed here as a general spatial PDE--RFDE centre-manifold theorem. Local Hopfs of the scalar core persist under the stated hypotheses. At yield parity the spectral gap vanishes and the vector system remains unreduced; that regime is the compensatory phase of Proposition~1 in~\cite{paperI}.
\end{abstract}

\section{Object}
\label{sec:object}

Let $i=1,\dots,n$ label specialised companion blocks $X^i=(N_i,A_i,Z_i,E_i)\in\mathbb R^4$, each with the institutional-failure vector field $F^i$ of~\cite{paperI}. Coupling is only through the service that enters the deficit. The $c$-th service of block $i$ is the companion's $C^\infty$ soft-minimum
\[
S_{i,c}=S_{i,c}^{\max}\Biggl[-\frac1{\rho}\ln\sum_m w_m\exp\Bigl(-\rho\frac{\tilde Y_{m,i,c}}{S_{i,c}^{\max}}\Bigr)\Biggr],
\]
with softmax weights $\pi_m=\partial S/\partial\tilde Y_m$. Write $X=(X^1,\dots,X^n)$ and
\begin{equation}
\dot X_t = F(X_t,X_{t-\tau})+\varepsilon_{\mathrm{phys}}\,R(X_t),
\label{eq:vec}
\end{equation}
where $F$ is the product of the $F^i$ with every occurrence of a lone-block yield replaced by the soft-minimum, and $R$ is a $C^1$ physical remainder (shared water, damage) of size one.

\begin{definition}[Uniform yield gap]
\label{def:gap}
Component $k$ maintains a uniform yield gap $\Delta_y>0$ on a set $\mathcal K$ if $\tilde Y_k\le \tilde Y_j-\Delta_y S^{\max}$ for every $j\neq k$, throughout $\mathcal K$.
\end{definition}

The companion's Lemma~17 gives $\pi_j\le w_{\min}^{-1}e^{-\rho\Delta_y}$ on $\mathcal K$, hence $\partial S/\partial N_j=O(e^{-\rho\Delta_y})$.

\section{Splitting}

Reorder as $X=(x,y)$ with $x=X^k$ the putative binding block and $y=(X^i)_{i\neq k}$ the slack blocks. Then
\begin{equation}
\begin{aligned}
\dot x &= F^k(x,x_\tau)+\varepsilon_c\,f(x,y,x_\tau,y_\tau),\\
\dot y &= G(y,y_\tau)+\varepsilon_c\,g(x,y,x_\tau,y_\tau),
\end{aligned}
\label{eq:split}
\end{equation}
where $\varepsilon_c=C e^{-\rho\Delta_y}+\varepsilon_{\mathrm{phys}}$ and $f,g$ are $C^1$ on a neighbourhood of a compact absorbing set, uniformly in the delay.

\begin{hypothesis}
\label{hyp:hyp}
There is a compact convex $\mathcal K_y$ on which the slack uncoupled system $\dot y=G(y,y_\tau)$ has a unique equilibrium $y_*$ that is exponentially stable, uniformly in histories in $C([-\tau,0],\mathcal K_y)$, with rate $\gamma_y>0$. The binding uncoupled system $\dot x=F^k(x,x_\tau)$ is $C^1$ and has a compact absorbing set $\mathcal K_x$.
\end{hypothesis}

Hypothesis~\ref{hyp:hyp} is the companion's delay-stable regime for every slack block (for example any block sitting in the monostable interval of its own $\tau$). It is an assumption on the slack fibres, not a claim that the binding block is stable.

\section{Attracting invariant graph}

\begin{theorem}[Yield-gap attracting invariant graph]
\label{thm:cm}
Assume Definition~\ref{def:gap} and Hypothesis~\ref{hyp:hyp}. There exist $\varepsilon_*$, $C$, and a $C^1$ map $h(\,\cdot\,;\varepsilon_c):C([-\tau,0],\mathcal K_x)\to C([-\tau,0],\mathcal K_y)$ with $h(\phi;0)\equiv y_*$, such that for every $\varepsilon_c\in[0,\varepsilon_*]$ the graph $y_t=h(x_t;\varepsilon_c)$ is locally invariant for~\eqref{eq:split} and exponentially attracting at rate $\gamma_y/2$. On that graph,
\begin{equation}
\dot x = F^k(x,x_\tau)+\varepsilon_c\,f\bigl(x,h(x;\varepsilon_c),x_\tau,h(x_\tau;\varepsilon_c)\bigr).
\label{eq:red}
\end{equation}
Every solution starting $O(\varepsilon_c)$-close to the graph satisfies
\[
\|y_t-h(x_t;\varepsilon_c)\|\le C e^{-\gamma_y t/2}\|y_0-h(x_0;\varepsilon_c)\|
\]
after a transient of length $O(|\log\varepsilon_c|)$.
\end{theorem}

\begin{proof}
At $\varepsilon_c=0$ the system~\eqref{eq:split} is triangular. The slack equation is the uncoupled RFDE $\dot y=G(y,y_\tau)$. By Hypothesis~\ref{hyp:hyp} this RFDE has a unique equilibrium $y_*$ that is exponentially stable, uniformly for histories in $C([-\tau,0],\mathcal K_y)$, with rate $\gamma_y>0$. The variational equation about $y_*$ therefore has spectrum in $\{\operatorname{Re}\lambda\le-\gamma_y\}$, so the fibre $\{x_t=\phi\}\times C([-\tau,0],\mathcal K_y)$ is normally hyperbolic, uniformly in the binding history $\phi\in C([-\tau,0],\mathcal K_x)$.

The perturbation of size $\varepsilon_c$ is $C^1$ on a neighbourhood of the compact set $\mathcal K_x\times\mathcal K_y$. The Hale--Lin invariant-manifold theorem for RFDEs \cite{hale1993} (equivalently, Fenichel's theorem for $C^1$ semiflows on Banach manifolds \cite{fenichel1979}) therefore yields, for all sufficiently small $\varepsilon_c$, a $C^1$ graph
\[
y_t=h(x_t;\varepsilon_c),\qquad h(\,\cdot\,;0)\equiv y_*,
\]
defined for $x_t$ in a neighbourhood of $C([-\tau,0],\mathcal K_x)$, locally invariant, and exponentially attracting at rate $\gamma_y/2$. Restricting~\eqref{eq:split} to the graph produces~\eqref{eq:red}. Solutions starting $O(\varepsilon_c)$-close to the graph approach it at rate $\gamma_y/2$ after a transient of length $O(|\log\varepsilon_c|)$, which is the standard Fenichel estimate for the distance to a normally hyperbolic manifold.
\end{proof}

This is a local attracting-invariant-graph statement for the weakly coupled RFDE under Hypothesis~\ref{hyp:hyp}; it is not asserted as a general spatial PDE--RFDE centre-manifold theorem. The companion's Theorem~18 is the $C^0([0,T])$ finite-time perturbation corollary obtained by treating slack blocks as $O(\varepsilon_c)$ inputs.

\begin{theorem}[Finite-time tracking of the scalar core]
\label{thm:track}
Let $x^0$ solve the companion scalar core $\dot x=F^k(x,x_\tau)$ and let $x^{\varepsilon_c}$ solve~\eqref{eq:red}. On every finite $[0,T]$,
\[
\|x^{\varepsilon_c}-x^0\|_{C([0,T])}\le C_T\varepsilon_c.
\]
\end{theorem}

\begin{proof}
The map $h(\,\cdot\,;\varepsilon_c)$ is $C^1$ and $h(\,\cdot\,;0)$ is constant, so
\[
f\bigl(x,h(x;\varepsilon_c),x_\tau,h(x_\tau;\varepsilon_c)\bigr)
\]
is $C^1$ and $O(1)$ on the absorbing set. The vector field of~\eqref{eq:red} is therefore a $C^1$ perturbation of $F^k$ of size $O(\varepsilon_c)$. Continuous dependence of RFDE solutions on the vector field \cite[Ch.~2]{hale1993} gives the $C^0([0,T])$ bound.
\end{proof}

\begin{theorem}[Hopf persistence]
\label{thm:hopf}
If the scalar core has a simple imaginary pair at $\tau_*$ with $\mathrm{Re}\,\lambda'(\tau_*)\neq 0$, then for every sufficiently small $\varepsilon_c$ the reduced equation~\eqref{eq:red} has a unique nearby Hopf at $\tau_*+O(\varepsilon_c)$.
\end{theorem}

\begin{proof}
Let $\Delta(\lambda,\tau)$ be the characteristic matrix of the scalar core $F^k$ at its interior rest, and let $\Delta^{\varepsilon_c}(\lambda,\tau)$ be that of~\eqref{eq:red}. Because $h(\,\cdot\,;\varepsilon_c)=y_*+O(\varepsilon_c)$ in $C^1$, one has $\Delta^{\varepsilon_c}=\Delta+O(\varepsilon_c)$ in the $C^1$ topology of matrix-valued quasi-polynomials. By hypothesis $\Delta(i\omega_*,\tau_*)=0$ is a simple imaginary root with $\partial_\tau\operatorname{Re}\lambda(\tau_*)\neq 0$. The implicit function theorem in the variables $(\omega,\tau)$, applied to $(\operatorname{Re},\operatorname{Im})\det\Delta^{\varepsilon_c}(i\omega,\tau)=0$, produces a unique nearby root $(\omega(\varepsilon_c),\tau(\varepsilon_c))$ with $\tau(\varepsilon_c)=\tau_*+O(\varepsilon_c)$.
\end{proof}

\section{Yield parity}

\begin{proposition}[Parity obstruction]
\label{prop:parity}
If $\Delta_y=0$ at a point of yield parity, the softmax weights are $O(1)$ and $\varepsilon_c$ is not exponentially small. Theorem~\ref{thm:cm} supplies no spectral gap on that set.
\end{proposition}

The reduced equation~\eqref{eq:red} therefore describes the terminal dynamics after a gap has opened. While substitutes remain comparable, the ledger stays the unreduced vector system of~\cite{paperI}.

\section{Conclusion}

Under a uniform yield gap and the stated slack-fibre spectral hypothesis, the weakly coupled vector Liebig RFDE admits a local $C^1$ attracting invariant graph over the binding block of~\cite{paperI}, with error $O(e^{-\rho\Delta_y})$. Local Hopfs of that block persist under the same hypotheses. At yield parity the gap hypothesis fails and the vector system is not reduced.
